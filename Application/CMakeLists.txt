cmake_minimum_required(VERSION 3.21)
project(RebasePortableDSP VERSION 0.1.0 LANGUAGES C)

# This lane builds the existing authored C kernels, not the desktop app.
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Portable DSP qualification currently requires Clang or AppleClang")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(REBASE_BUILD_SOXR "Link the wrapper to an already-built pinned SoXR library" OFF)
set(REBASE_SOXR_LIBRARY "" CACHE FILEPATH "Pinned SoXR library when REBASE_BUILD_SOXR is ON")

add_library(rebase_strict_fp INTERFACE)
if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(rebase_strict_fp INTERFACE /fp:strict /clang:-fno-fast-math /clang:-ffp-contract=off /clang:-frounding-math)
else()
  target_compile_options(rebase_strict_fp INTERFACE -fno-fast-math -ffp-contract=off -frounding-math)
endif()

function(rebase_kernel target directory)
  file(GLOB sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Sources/${directory}/*.c")
  add_library(${target} STATIC ${sources})
  target_include_directories(${target} PUBLIC "Sources/${directory}/include")
  target_link_libraries(${target} PUBLIC rebase_strict_fp)
  if(NOT WIN32)
    target_link_libraries(${target} PUBLIC m)
  endif()
endfunction()
rebase_kernel(RebaseDSP RebaseDSP)
rebase_kernel(RebaseAmbiDSP RebaseAmbiDSP)
rebase_kernel(RebaseTimingDSP RebaseTimingDSP)
if(REBASE_BUILD_SOXR)
  if(NOT EXISTS "${REBASE_SOXR_LIBRARY}")
    message(FATAL_ERROR "Set REBASE_SOXR_LIBRARY to the existing pinned SoXR build")
  endif()
  rebase_kernel(RebaseSoXR RebaseSoXR)
  target_link_libraries(RebaseSoXR PUBLIC RebaseTimingDSP "${REBASE_SOXR_LIBRARY}")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(rebase_portable_dsp_tests Tests/PortableDSP/conformance.c)
  target_link_libraries(rebase_portable_dsp_tests PRIVATE RebaseDSP RebaseAmbiDSP RebaseTimingDSP)
  add_test(NAME portable_dsp_conformance COMMAND rebase_portable_dsp_tests)
endif()
