cmake_minimum_required(VERSION 3.24)
project(RebaseWindowsAudio VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(REBASE_RUN_AUDIO_DEVICE_TESTS "Opt in to audible tests on a real Windows endpoint" OFF)
set(REBASE_SOXR_LIBRARY "" CACHE FILEPATH "Already-built pinned public SoXR 0.1.3 library or import library")
set(REBASE_SOXR_RUNTIME "" CACHE FILEPATH "Pinned public SoXR runtime DLL to copy beside Windows executables")
if(NOT EXISTS "${REBASE_SOXR_LIBRARY}")
  message(FATAL_ERROR "Set REBASE_SOXR_LIBRARY to the verified pinned public SoXR build")
endif()

add_library(rebase_audio_lifecycle STATIC src/PlaybackCursor.cpp src/RatePlaybackCursor.cpp)
target_include_directories(rebase_audio_lifecycle PUBLIC include src)
add_library(rebase_audio_conversion STATIC src/RateConversion.cpp)
target_include_directories(rebase_audio_conversion PUBLIC include src ../../ThirdParty/soxr-0.1.3/src)
target_link_libraries(rebase_audio_conversion PUBLIC rebase_audio_lifecycle "${REBASE_SOXR_LIBRARY}")
if(MSVC)
  target_compile_options(rebase_audio_lifecycle PRIVATE /W4 /WX /fp:strict)
  target_compile_options(rebase_audio_conversion PRIVATE /W4 /WX /EHsc /fp:strict)
else()
  target_compile_options(rebase_audio_lifecycle PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
  target_compile_options(rebase_audio_conversion PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()

if(BUILD_TESTING)
  add_executable(rebase_audio_lifecycle_tests tests/lifecycle.cpp)
  target_link_libraries(rebase_audio_lifecycle_tests PRIVATE rebase_audio_lifecycle)
  add_test(NAME native_audio_portable_lifecycle COMMAND rebase_audio_lifecycle_tests)
  add_executable(rebase_audio_conversion_tests tests/rate_conversion.cpp)
  target_link_libraries(rebase_audio_conversion_tests PRIVATE rebase_audio_conversion)
  add_test(NAME native_audio_rate_conversion COMMAND rebase_audio_conversion_tests)
endif()

if(WIN32)
  add_library(RebaseWindowsAudio SHARED src/WasapiOutput.cpp)
  target_include_directories(RebaseWindowsAudio PUBLIC include PRIVATE src)
  target_compile_definitions(RebaseWindowsAudio PRIVATE REBASE_AUDIO_BUILD _WIN32_WINNT=0x0A00)
  target_link_libraries(RebaseWindowsAudio PRIVATE rebase_audio_conversion ole32 uuid ksuser)
  if(MSVC)
    target_compile_options(RebaseWindowsAudio PRIVATE /W4 /WX /EHsc /fp:strict)
  else()
    target_compile_options(RebaseWindowsAudio PRIVATE -Wall -Wextra -Werror -fno-fast-math)
  endif()
  add_executable(rebase_wasapi_probe tests/wasapi_probe.cpp)
  target_link_libraries(rebase_wasapi_probe PRIVATE RebaseWindowsAudio)
  if(NOT EXISTS "${REBASE_SOXR_RUNTIME}")
    message(FATAL_ERROR "Set REBASE_SOXR_RUNTIME to the matching pinned soxr.dll")
  endif()
  add_custom_command(TARGET RebaseWindowsAudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${REBASE_SOXR_RUNTIME}" "$<TARGET_FILE_DIR:RebaseWindowsAudio>")
  if(BUILD_TESTING)
    add_custom_command(TARGET rebase_audio_conversion_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${REBASE_SOXR_RUNTIME}" "$<TARGET_FILE_DIR:rebase_audio_conversion_tests>")
  endif()
  if(BUILD_TESTING AND REBASE_RUN_AUDIO_DEVICE_TESTS)
    add_test(NAME native_audio_device_lifecycle COMMAND rebase_wasapi_probe --allow-audio)
    set_tests_properties(native_audio_device_lifecycle PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  endif()
endif()
